cmake_minimum_required(VERSION 3.16)
project(OrcaLanMonitorProbe LANGUAGES CXX)
find_package(Threads REQUIRED)

add_executable(lan_monitor_probe main.cpp)
target_include_directories(lan_monitor_probe PRIVATE ../../src)
add_executable(lan_monitor_worker ../../src/slic3r/Utils/LanMonitorWorker.cpp)
add_executable(lan_camera_probe ../../src/dev-utils/lan_camera_probe.cpp)
target_include_directories(lan_camera_probe PRIVATE ../../deps_src ../../src)
target_link_libraries(lan_camera_probe PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(lan_camera_probe PRIVATE ws2_32)
endif()
target_include_directories(lan_monitor_worker PRIVATE ../../deps_src ../../src)
target_link_libraries(lan_monitor_worker PRIVATE Threads::Threads)

foreach(target lan_monitor_probe lan_monitor_worker lan_camera_probe)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

option(LAN_MONITOR_BUILD_TESTS "Build the focused LAN snapshot Catch2 tests" OFF)
if(LAN_MONITOR_BUILD_TESTS)
    enable_testing()
    add_subdirectory(../../tests/catch2 catch2 EXCLUDE_FROM_ALL)
    add_executable(lan_monitor_snapshot_tests ../../tests/slic3rutils/test_lan_monitor.cpp)
    target_compile_features(lan_monitor_snapshot_tests PRIVATE cxx_std_17)
    target_include_directories(lan_monitor_snapshot_tests PRIVATE ../../src ../../deps_src)
    target_link_libraries(lan_monitor_snapshot_tests PRIVATE Catch2::Catch2WithMain)
    add_test(NAME lan_monitor_snapshots COMMAND lan_monitor_snapshot_tests "[LanMonitor]" --order rand)
endif()
