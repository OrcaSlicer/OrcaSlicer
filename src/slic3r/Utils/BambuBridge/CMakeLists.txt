# Standalone build: this executable must match the DLL, independently of Orca's architecture.
cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(OrcaBambuNetworkHelper LANGUAGES CXX)

get_filename_component(ORCA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)

option(BAMBU_BRIDGE_WINDOWS_SMOKE "Build the fake DLL and process-boundary smoke test" OFF)
option(BAMBU_BRIDGE_TESTS_ONLY "Build portable RPC tests without the Windows helper" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Bambu's release DLL uses the release MSVC STL ABI; build the helper as Release too.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

if (BAMBU_BRIDGE_TESTS_ONLY)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(bambu_bridge_tests
        "${ORCA_ROOT}/tests/slic3rutils/test_bambu_bridge.cpp"
        "${ORCA_ROOT}/tests/catch2/extras/catch_amalgamated.cpp")
    target_include_directories(bambu_bridge_tests PRIVATE "${ORCA_ROOT}/src" "${ORCA_ROOT}/deps_src"
        "${ORCA_ROOT}/tests/catch2/extras")
    target_compile_definitions(bambu_bridge_tests PRIVATE BAMBU_BRIDGE_STANDALONE_TESTS)
    target_link_libraries(bambu_bridge_tests PRIVATE Threads::Threads)
    add_test(NAME bambu_bridge_tests COMMAND bambu_bridge_tests)
else ()
    if (NOT WIN32)
        message(FATAL_ERROR "Build the helper with a Windows x64 or Win32 toolchain")
    endif ()
    if (MSVC_CXX_ARCHITECTURE_ID MATCHES "ARM")
        message(FATAL_ERROR "The helper must match the foreign DLL: use -A x64 or -A Win32, not ARM64")
    endif ()
    add_executable(orca-bambu-network-helper HelperMain.cpp WinPipe.cpp)
    target_include_directories(orca-bambu-network-helper PRIVATE "${ORCA_ROOT}/src" "${ORCA_ROOT}/deps_src")
    target_compile_definitions(orca-bambu-network-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
    target_link_libraries(orca-bambu-network-helper PRIVATE bcrypt advapi32)
    if (MINGW)
        target_link_options(orca-bambu-network-helper PRIVATE -municode)
    endif ()
    install(TARGETS orca-bambu-network-helper RUNTIME DESTINATION .)
    if (BAMBU_BRIDGE_WINDOWS_SMOKE)
        enable_testing()
        add_library(bambu-bridge-fake SHARED "${ORCA_ROOT}/tests/slic3rutils/bambu_bridge_fake_dll.cpp")
        add_executable(bambu-bridge-smoke Client.cpp WinPipe.cpp "${ORCA_ROOT}/tests/slic3rutils/bambu_bridge_windows_smoke.cpp")
        foreach(target bambu-bridge-fake bambu-bridge-smoke)
            target_include_directories(${target} PRIVATE "${ORCA_ROOT}/src" "${ORCA_ROOT}/deps_src")
            target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
        endforeach()
        target_link_libraries(bambu-bridge-smoke PRIVATE bcrypt advapi32)
        if (MINGW)
            target_link_options(bambu-bridge-smoke PRIVATE -municode)
        endif ()
        add_test(NAME bambu_bridge_process_smoke COMMAND bambu-bridge-smoke
            "$<TARGET_FILE:orca-bambu-network-helper>" "$<TARGET_FILE:bambu-bridge-fake>")
        set_tests_properties(bambu_bridge_process_smoke PROPERTIES TIMEOUT 60)
    endif ()
endif ()
